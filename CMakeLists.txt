cmake_minimum_required(VERSION 3.13)
project(HW2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
add_executable(func-analyzer FuncAnalyzer/FuncAnalyzer.cpp)
add_executable(strength-reduction StrengthReduction/StrengthReduction.cpp)
foreach(tool func-analyzer strength-reduction)
  if(NOT LLVM_ENABLE_RTTI)
    target_compile_options(${tool} PRIVATE -fno-rtti)
  endif()
  if(TARGET clang-cpp AND TARGET LLVM)
    target_link_libraries(${tool} PRIVATE clang-cpp LLVM)
  else()
    llvm_map_components_to_libnames(llvm_libs support)
    target_link_libraries(${tool} PRIVATE clangTooling clangFrontend clangRewrite clangAST clangLex clangBasic ${llvm_libs})
  endif()
endforeach()
